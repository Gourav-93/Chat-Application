using System.Diagnostics;
using System.Net;
using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Net.Sockets;
using System.Net.WebSockets;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using Chat.Application.DTOs;
using Chat.Application.Interfaces;
using Chat.Application.Services;
using Chat.Domain.Entities;
using Chat.Infrastructure.Data;
using Microsoft.Data.SqlClient;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;

var root = args[0];
var config = new ConfigurationBuilder().SetBasePath(root).AddJsonFile("appsettings.json").Build();
var connectionString = config.GetConnectionString("DefaultConnection")!;
var target = new SqlConnectionStringBuilder(connectionString);
if (target.DataSource != "localhost" || target.InitialCatalog != "ChatDb")
    throw new Exception("Only confirmed localhost/ChatDb is supported.");
config["Otp:HashingKey"] = Convert.ToHexString(RandomNumberGenerator.GetBytes(32));
var jwtKey = Convert.ToHexString(RandomNumberGenerator.GetBytes(32));
ChatDbContext Db() => new(new DbContextOptionsBuilder<ChatDbContext>().UseSqlServer(connectionString).Options);
var sender = new CapturingSender();
var prefix = "otp_check_" + Guid.NewGuid().ToString("N")[..10];
string Mobile() => "+999" + RandomNumberGenerator.GetInt32(100000000, 999999999);
void Check(bool condition, string name)
{
    if (!condition) throw new Exception("FAIL: " + name);
    Console.WriteLine("PASS: " + name);
}
async Task<(int Status, OtpVerificationResult? Result)> Verify(string mobile, string otp)
{
    await using var db = Db();
    try { return (200, await new OtpService(db, sender, config).VerifyOtpAsync(new() { MobileNumber = mobile, Otp = otp })); }
    catch (AuthFlowException ex) { return (ex.StatusCode, null); }
}
async Task<(int Status, Guid Id)> Complete(string token, string name)
{
    await using var db = Db();
    try { return (200, await new OtpService(db, sender, config).CompleteProfileAsync(new() { RegistrationToken = token, UserName = name })); }
    catch (AuthFlowException ex) { return (ex.StatusCode, Guid.Empty); }
    catch (ArgumentException) { return (400, Guid.Empty); }
}
async Task<string> Seed(string mobile)
{
    var otp = RandomNumberGenerator.GetInt32(100000, 1000000).ToString();
    using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(config["Otp:HashingKey"]!));
    await using var db = Db();
    db.OtpVerifications.Add(new() { MobileNumber = mobile,
        OtpHash = Convert.ToHexString(hmac.ComputeHash(Encoding.UTF8.GetBytes($"{mobile}:{otp}"))),
        CreatedAt = DateTime.UtcNow, ExpiresAt = DateTime.UtcNow.AddMinutes(5) });
    await db.SaveChangesAsync();
    return otp;
}
try
{
    await using (var db = Db())
        Check(!(await db.Database.GetPendingMigrationsAsync()).Any(), "All migrations applied, including pending registration table");
    var mobile = Mobile();
    await using (var db = Db())
        await new OtpService(db, sender, config).SendOtpAsync(new() { MobileNumber = " " + mobile + " " });
    var otp = sender.LastCode!;
    await using (var db = Db())
        Check(await db.OtpVerifications.CountAsync(o => o.MobileNumber == mobile && o.OtpHash != otp) == 1 && !await db.Users.AnyAsync(u => u.MobileNumber == mobile), "Send persists normalized hashed OTP without user creation");
    var race = await Task.WhenAll(Verify(mobile, otp), Verify(" " + mobile + " ", otp));
    Check(race.Count(r => r.Status == 200) == 1 && race.Count(r => r.Status == 400) == 1, "Parallel OTP verification consumes challenge once");
    var pendingResult = race.Single(r => r.Status == 200).Result!;
    var credential = pendingResult.RegistrationToken!;
    Guid pendingId;
    await using (var db = Db())
    {
        var pending = await db.PendingRegistrations.SingleAsync(p => p.MobileNumber == mobile);
        pendingId = pending.Id;
        Check(pendingResult.UserId == null && !await db.Users.AnyAsync(u => u.MobileNumber == mobile), "New verified mobile stays pending without a User");
        Check(pending.TokenHash.Length == 64 && pending.TokenHash != credential[33..] && pending.CompletedAt == null, "Only registration credential hash is persisted");
    }
    Check((await Complete(credential, " ")).Status == 400, "Blank name rejected without consuming registration");
    var tampered = credential[..^1] + (credential[^1] == '0' ? "1" : "0");
    Check((await Complete(tampered, prefix)).Status == 401, "Tampered registration credential rejected");
    await using (var db = Db())
    {
        (await db.PendingRegistrations.SingleAsync(p => p.Id == pendingId)).ExpiresAt = DateTime.UtcNow.AddSeconds(-1);
        await db.SaveChangesAsync();
    }
    Check((await Complete(credential, prefix)).Status == 410, "Expired registration credential rejected");
    var resumed = (await Verify(mobile, await Seed(mobile))).Result!;
    Check(resumed.RegistrationToken![..32] == credential[..32] && resumed.RegistrationToken != credential, "Fresh OTP resumes same pending row and rotates credential");
    Check((await Complete(credential, prefix)).Status == 401, "Rotated old credential cannot complete profile");
    var completions = await Task.WhenAll(Complete(resumed.RegistrationToken!, " " + prefix + " "), Complete(resumed.RegistrationToken!, prefix));
    Check(completions.Count(c => c.Status == 200) == 1 && completions.Count(c => c.Status == 409) == 1, "Concurrent profile submissions create one account and consume token once");
    var userId = completions.Single(c => c.Status == 200).Id;
    Check((await Complete(resumed.RegistrationToken!, prefix + "_overwrite")).Status == 409, "Completed registration token cannot replay or rename");
    var existing = (await Verify(mobile, await Seed(mobile))).Result!;
    Check(existing.UserId == userId && existing.RegistrationToken == null, "Existing mobile signs in directly without a name or profile token");
    await using (var db = Db())
    {
        Check(await db.PendingRegistrations.CountAsync(p => p.MobileNumber == mobile) == 1 &&
            await db.Users.CountAsync(u => u.MobileNumber == mobile) == 1 &&
            (await db.Users.SingleAsync(u => u.Id == userId)).UserName == prefix, "Retry retains one pending row, one user and original trimmed name");
        db.Users.Add(new() { Id = Guid.NewGuid(), UserName = prefix + "_duplicate", MobileNumber = mobile });
        try { await db.SaveChangesAsync(); Check(false, "Unique mobile index"); }
        catch (DbUpdateException ex) when (ex.InnerException is SqlException sql && (sql.Number == 2601 || sql.Number == 2627))
        { Check(true, "Database unique mobile constraint rejects bypassing writer"); }
    }

    var listener = new TcpListener(IPAddress.Loopback, 0); listener.Start();
    var port = ((IPEndPoint)listener.LocalEndpoint).Port; listener.Stop();
    var start = new ProcessStartInfo("dotnet") { WorkingDirectory = root, UseShellExecute = false, CreateNoWindow = true, RedirectStandardOutput = true, RedirectStandardError = true };
    start.ArgumentList.Add(Path.Combine(root, "bin/Debug/net10.0/Two-Way-Com.dll"));
    start.Environment["ASPNETCORE_URLS"] = $"http://127.0.0.1:{port}";
    start.Environment["ASPNETCORE_ENVIRONMENT"] = "Development";
    start.Environment["ConnectionStrings__DefaultConnection"] = connectionString;
    start.Environment["Otp__HashingKey"] = config["Otp:HashingKey"];
    start.Environment["Jwt__Key"] = jwtKey;
    start.Environment["Jwt__Issuer"] = "otp-checks";
    start.Environment["Jwt__Audience"] = "otp-checks";
    start.Environment["Logging__LogLevel__Default"] = "Error";
    Process StartServer()
    {
        var process = Process.Start(start)!;
        _ = process.StandardOutput.ReadToEndAsync(); _ = process.StandardError.ReadToEndAsync();
        return process;
    }
    var server = StartServer();
    try
    {
        using var client = new HttpClient { BaseAddress = new Uri($"http://127.0.0.1:{port}") };
        async Task Ready()
        {
            for (var i = 0; i < 100; i++)
            {
                try { using var response = await client.GetAsync("/api/conversations"); return; }
                catch (HttpRequestException) { await Task.Delay(100); }
            }
            Check(false, "HTTP server starts");
        }
        await Ready();
        var html = await client.GetStringAsync("/index.html");
        var initialForm = html.Split("<form id=\"loginForm\">")[1].Split("</form>")[0];
        Check(!initialForm.Contains("userName") && html.Contains("id=\"profileForm\""), "Served UI separates mobile/OTP and name form");
        var js = await client.GetStringAsync("/js/chat.js");
        Check(js.Contains("JSON.stringify({ mobileNumber, otp })") && js.Contains("showPendingRegistration(pending)"), "Test UI verifies without name and restores pending session");
        foreach (var path in new[] { "/api/auth/login", "/api/auth/register" })
            Check((await client.PostAsJsonAsync(path, new { })).StatusCode == HttpStatusCode.NotFound, "Removed password route: " + path);
        var httpMobile = Mobile();
        var verifyResponse = await client.PostAsJsonAsync("/api/auth/verify-otp", new { mobileNumber = httpMobile, otp = await Seed(httpMobile) });
        var verification = await verifyResponse.Content.ReadFromJsonAsync<JsonElement>();
        Check(verifyResponse.IsSuccessStatusCode && verification.GetProperty("status").GetString() == "profileRequired" &&
            !verification.TryGetProperty("login", out _) && !verification.TryGetProperty("token", out _), "New HTTP verification grants no login JWT");
        var registration = verification.GetProperty("registrationToken").GetString()!;
        client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", registration);
        Check((await client.GetAsync("/api/conversations")).StatusCode == HttpStatusCode.Unauthorized, "Registration token cannot access protected HTTP chat API");
        Check((await client.PostAsync("/hubs/chat/negotiate?negotiateVersion=1", null)).StatusCode == HttpStatusCode.Unauthorized, "Registration token cannot authenticate SignalR");
        client.DefaultRequestHeaders.Authorization = null;
        Check((await client.PostAsJsonAsync("/api/auth/complete-profile", new { userName = prefix, mobileNumber = httpMobile, isNewUser = true })).StatusCode == HttpStatusCode.Unauthorized, "Mobile and isNewUser flag cannot authorize registration");
        await using (var db = Db())
            Check(!await db.Users.AnyAsync(u => u.MobileNumber == httpMobile), "Unauthorized completion leaves no user");

        server.Kill(entireProcessTree: true); await server.WaitForExitAsync(); server.Dispose();
        server = StartServer(); await Ready();
        var otherMobile = Mobile();
        var completedResponse = await client.PostAsJsonAsync("/api/auth/complete-profile", new
        { userName = prefix + "_http", registrationToken = registration, mobileNumber = otherMobile, isNewUser = false });
        Check(completedResponse.IsSuccessStatusCode, "Pending registration resumes after backend restart");
        var first = (await completedResponse.Content.ReadFromJsonAsync<LoginResponseDto>())!;
        Check(first.MobileNumber == httpMobile && first.UserName == prefix + "_http", "Completion uses token-bound mobile and entered name, ignoring client mobile/flag");
        Check((await client.PostAsJsonAsync("/api/auth/complete-profile", new { userName = prefix + "_changed", registrationToken = registration })).StatusCode == HttpStatusCode.Conflict, "HTTP completion replay rejected");
        async Task<ClientWebSocket> Connect(string token)
        {
            var socket = new ClientWebSocket();
            await socket.ConnectAsync(new Uri($"ws://127.0.0.1:{port}/hubs/chat?access_token={Uri.EscapeDataString(token)}"), CancellationToken.None);
            await socket.SendAsync(Encoding.UTF8.GetBytes("{\"protocol\":\"json\",\"version\":1}\u001e"), WebSocketMessageType.Text, true, CancellationToken.None);
            var buffer = new byte[8192];
            using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(10));
            var read = await socket.ReceiveAsync(new ArraySegment<byte>(buffer), timeout.Token);
            Check(read.MessageType == WebSocketMessageType.Text && Encoding.UTF8.GetString(buffer, 0, read.Count).StartsWith("{}"), "Authenticated SignalR handshake accepted");
            return socket;
        }
        using var socket1 = await Connect(first.Token);
        await Task.Delay(1100);
        var loginResponse = await client.PostAsJsonAsync("/api/auth/verify-otp", new { mobileNumber = httpMobile, otp = await Seed(httpMobile), userName = prefix + "_ignored" });
        var loginJson = await loginResponse.Content.ReadFromJsonAsync<JsonElement>();
        Check(loginResponse.IsSuccessStatusCode && loginJson.GetProperty("status").GetString() == "signedIn", "Existing user receives direct sign-in response");
        var second = loginJson.GetProperty("login").Deserialize<LoginResponseDto>(new JsonSerializerOptions(JsonSerializerDefaults.Web))!;
        Check(second.UserId == first.UserId && second.UserName == first.UserName && first.Token != second.Token, "Existing name preserved and independent JWT issued");
        foreach (var login in new[] { first, second })
        {
            client.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", login.Token);
            Check((await client.GetAsync("/api/conversations")).IsSuccessStatusCode, "Both JWTs remain accepted after second login");
        }
        using var socket2 = await Connect(second.Token);
        var probe = $"{{\"type\":1,\"invocationId\":\"still-connected\",\"target\":\"GetUserStatus\",\"arguments\":[\"{first.UserId}\"]}}\u001e";
        await socket1.SendAsync(Encoding.UTF8.GetBytes(probe), WebSocketMessageType.Text, true, CancellationToken.None);
        var probeBuffer = new byte[8192];
        using var probeTimeout = new CancellationTokenSource(TimeSpan.FromSeconds(10));
        var probeResult = await socket1.ReceiveAsync(new ArraySegment<byte>(probeBuffer), probeTimeout.Token);
        Check(Encoding.UTF8.GetString(probeBuffer, 0, probeResult.Count).Contains("\"invocationId\":\"still-connected\""), "Device one hub responds after device two logs in");
        Check(socket1.State == WebSocketState.Open && socket2.State == WebSocketState.Open, "Two devices remain connected");
        await using var final = Db();
        Check(await final.Users.CountAsync(u => u.MobileNumber == httpMobile) == 1 && !await final.Users.AnyAsync(u => u.MobileNumber == otherMobile), "One account remains on the server-authorized mobile");
    }
    finally { if (!server.HasExited) { server.Kill(entireProcessTree: true); await server.WaitForExitAsync(); } server.Dispose(); }
    Console.WriteLine("All checks passed. Retained fixture prefix: " + prefix);
}
catch (Exception ex)
{
    Console.Error.WriteLine("Checks failed: " + ex.GetType().Name + (ex.Message.StartsWith("FAIL:") ? " " + ex.Message : "; inspect locally without exposing secrets."));
    Environment.ExitCode = 1;
}
sealed class CapturingSender : ISmsSender
{
    public string? LastCode { get; private set; }
    public Task SendOtpAsync(string mobileNumber, string otp) { LastCode = otp; return Task.CompletedTask; }
}
